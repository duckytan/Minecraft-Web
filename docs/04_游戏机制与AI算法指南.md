## 1. 游戏核心机制

### 1.1 玩法循环（Game Loop）

```
[玩家输入] → [位置更新] → [碰撞检测] → [世界更新] → [渲染] → 继续循环
```

### 1.2 基础规则

| 机制 | 规则 |
| --- | --- |
| 移动速度 | 5 单位/秒 |
| 奔跑倍率 | 2x |
| 跳跃高度 | 3 单位 |
| 重力加速度 | 9.8 单位/秒² |
| 交互距离 | 5 单位（Raycasting 范围） |

### 1.3 方块放置/破坏规则

- **破坏**：左键点击 → Raycasting 找到目标方块 → 移除
- **放置**：右键点击 → 目标方块相邻面放置 → 碰撞检测（不与玩家重叠）

function generateFlatTerrain(chunkX, chunkZ) {
  const blocks = [];
  for (let x = 0; x < 16; x++) {
    for (let z = 0; z < 16; z++) {
      for (let y = 0; y < 4; y++) {

### 2.2 Perlin/Simplex 噪声地形

```javascript
import { createNoise2D } from 'simplex-noise';

const noise = createNoise2D();

function generateNoiseHeight(x, z) {
  const scale = 0.05;     // 控制频率
  const amplitude = 10;   // 控制幅度
  const baseHeight = 10;  // 基准高度
  
  const value = noise(x * scale, z * scale);
  return Math.floor(baseHeight + value * amplitude);
}

function generateNoiseTerrain(chunkX, chunkZ) {
  const blocks = [];
  for (let x = 0; x < 16; x++) {
    for (let z = 0; z < 16; z++) {
      const worldX = chunkX * 16 + x;
      const worldZ = chunkZ * 16 + z;
      const height = generateNoiseHeight(worldX, worldZ);
      
      for (let y = 0; y <= height; y++) {
        let type;
        if (y === height) type = 'grass';
        else if (y >= height - 3) type = 'dirt';
        else type = 'stone';
        
        blocks.push({ position: [x, y, z], type });
      }
    }
  }
  return blocks;
}
```

function getBiome(x, z) {
  const humidity = noise(x * 0.01, z * 0.01);
  const temperature = noise((x + 1000) * 0.01, z * 0.01);
  
  if (humidity > 0.3 && temperature > 0.3) return 'jungle';

## 4. 物理与碰撞检测

### 4.1 AABB 碰撞检测

```javascript
class BoundingBox {
  constructor(public min: Vector3, public max: Vector3) {}
  
  intersects(other: BoundingBox): boolean {
    return (
      this.min.x <= other.max.x && this.max.x >= other.min.x &&
      this.min.y <= other.max.y && this.max.y >= other.min.y &&
      this.min.z <= other.max.z && this.max.z >= other.min.z
    );
  }
}

function checkCollisionWithWorld(playerBox: BoundingBox, world: World): boolean {
  // 获取玩家周围的方块
  const nearbyBlocks = world.getBlocksInRange(playerBox);
  for (const block of nearbyBlocks) {
    if (block.solid && playerBox.intersects(block.boundingBox)) {
      return true;
    }
  }
  return false;
}
```

### 4.2 重力与跳跃

```javascript
class Player {
  velocity = new Vector3(0, 0, 0);
  onGround = false;
  
  update(delta: number) {
    // 应用重力
    this.velocity.y -= GRAVITY * delta;
    
    // 应用速度
    this.position.add(this.velocity.clone().multiplyScalar(delta));
    
    // 碰撞检测
    if (checkCollisionWithWorld(this.getBoundingBox(), this.world)) {
      this.velocity.y = 0;
      this.onGround = true;
    } else {
      this.onGround = false;
    }
  }
  
  jump() {
    if (this.onGround) {
      this.velocity.y = JUMP_VELOCITY;
    }
  }
}
```

## 6. 性能优化算法

### 6.1 面剔除（Greedy Meshing）

```javascript
function shouldRenderFace(
  world: World, 
  blockPos: Vector3, 
  faceDirection: Vector3
): boolean {
  const neighborPos = blockPos.clone().add(faceDirection);
  const neighborBlock = world.getBlock(neighborPos);
  
  // 如果邻居是透明方块或空气，则渲染该面
  return !neighborBlock || neighborBlock.transparent;
}
```

### 6.2 Chunk 视锥剔除

```javascript
function isChunkVisible(chunk: Chunk, camera: Camera): boolean {
  const frustum = new THREE.Frustum();
  frustum.setFromProjectionMatrix(
    camera.projectionMatrix.clone().multiply(camera.matrixWorldInverse)
  );
  return frustum.intersectsBox(chunk.boundingBox);
}
```

### 7.1 主游戏循环

```javascript
class Game {
  private lastTime = 0;
  
  init() {
    this.scene = createScene();
    this.world = new World();
    this.player = new Player();
    this.renderer = createRenderer();
    this.start();
  }
  
  start() {
    requestAnimationFrame(this.loop.bind(this));
  }
  
  loop(currentTime: number) {
    const delta = (currentTime - this.lastTime) / 1000; // 秒
    this.lastTime = currentTime;
    
    this.handleInput();
    this.player.update(delta);
    this.world.update(delta);
    this.renderer.render(this.scene, this.player.camera);
    
    requestAnimationFrame(this.loop.bind(this));
  }

  }
}
```

### 7.2 Raycasting 示例

```javascript
function performRaycast(camera: Camera, mouse: Vector2): { block: Block; face: Vector3 } | null {
  const raycaster = new THREE.Raycaster();
  raycaster.setFromCamera(mouse, camera);
  
  const intersections = raycaster.intersectObjects(world.getAllMeshes());
  if (intersections.length > 0) {
    const hit = intersections[0];
    const block = world.getBlockAt(hit.point);
    const face = hit.face.normal;
    return { block, face };
  }
  return null;
}
```
